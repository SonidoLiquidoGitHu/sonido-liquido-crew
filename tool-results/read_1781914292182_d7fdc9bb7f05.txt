   455→export async function POST() {
   456→  try {
   457→    if (!isDatabaseConfigured()) {
   458→      return NextResponse.json(
   459→        { success: false, error: "Database not configured" },
   460→        { status: 503 }
   461→      );
   462→    }
   463→
   464→    const connected = await checkConnection();
   465→    if (!connected) {
   466→      return NextResponse.json(
   467→        { success: false, error: "Database connection failed" },
   468→        { status: 503 }
   469→      );
   470→    }
   471→
   472→    const results: { table: string; status: string; error?: string }[] = [];
   473→
   474→    // Create tables
   475→    for (const sql of ENSURE_TABLES_SQL) {
   476→      const tableNameMatch = sql.match(/CREATE TABLE IF NOT EXISTS (\w+)/);
   477→      const tableName = tableNameMatch ? tableNameMatch[1] : "unknown";
   478→
   479→      try {
   480→        await executeRaw(sql);
   481→        results.push({ table: tableName, status: "ok" });
   482→      } catch (error) {
   483→        const msg = error instanceof Error ? error.message : String(error);
   484→        results.push({ table: tableName, status: "error", error: msg });
   485→      }
   486→    }
   487→
   488→    // Add missing columns to existing tables (safe - ignores "duplicate column" errors)
   489→    for (const sql of ADD_COLUMNS_SQL) {
   490→      const colMatch = sql.match(/ALTER TABLE (\w+) ADD COLUMN (\w+)/);
   491→      const tableName = colMatch ? colMatch[1] : "unknown";
   492→      const colName = colMatch ? colMatch[2] : "unknown";
   493→
   494→      try {
   495→        await executeRaw(sql);
   496→        results.push({ table: `${tableName}.${colName}`, status: "added" });
   497→      } catch (error) {
   498→        const msg = error instanceof Error ? error.message : String(error);
   499→        if (msg.includes("duplicate column name")) {
   500→          results.push({ table: `${tableName}.${colName}`, status: "exists" });
   501→        } else {
   502→          results.push({ table: `${tableName}.${colName}`, status: "error", error: msg });
   503→        }
   504→      }
   505→    }
   506→
   507→    // Create indexes
   508→    for (const sql of ENSURE_INDEXES_SQL) {
   509→      try {
   510→        await executeRaw(sql);
   511→      } catch (error) {
   512→        console.warn("[Ensure Tables] Index creation warning:", error);
   513→      }
   514→    }
   515→
   516→    // Apply data fixes
   517→    for (const sql of DATA_FIXES_SQL) {
   518→      try {
   519→        await executeRaw(sql);
   520→        results.push({ table: "data_fix", status: "applied", error: sql.substring(0, 80) });
   521→      } catch (error) {
   522→        const msg = error instanceof Error ? error.message : String(error);
   523→        results.push({ table: "data_fix", status: "error", error: msg });
   524→      }
   525→    }
   526→
   527→    // === SEED ARTIST EXTERNAL PROFILES ===
   528→    try {
   529→      // Check how many artists already have profiles using drizzle
   530→      const [profileCountRow] = await db
   531→        .select({ count: sql<number>`COUNT(DISTINCT ${artistExternalProfiles.artistId})` })
   532→        .from(artistExternalProfiles);
   533→      const profileCount = profileCountRow?.count ?? 0;
   534→
   535→      // Get total active artists
   536→      const [artistCountRow] = await db
   537→        .select({ count: sql<number>`COUNT(*)` })
   538→        .from(artists)
   539→        .where(eq(artists.isActive, true));
   540→      const artistCount = artistCountRow?.count ?? 0;
   541→
   542→      // If some artists don't have profiles, seed them
   543→      if (profileCount < artistCount) {
   544→        // Read social URLs from JSON file
   545→        const socialUrlsPath = join(process.cwd(), "data", "artist-social-urls.json");
   546→        let socialData: Record<string, {
   547→          name?: string;
   548→          spotify?: string;
   549→          spotifyId?: string;
   550→          instagram?: string | null;
   551→        }> = {};
   552→        try {
   553→          const fileContents = readFileSync(socialUrlsPath, "utf-8");
   554→          socialData = JSON.parse(fileContents).artists || {};
   555→        } catch {
